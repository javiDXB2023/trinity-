recognition.continuous = false;
recognition.lang = 'en-US';

MATRIX EFFECT 
#!/bin/sh

CHARS="ｱｲｳｴｵ01JARVIS"
COLS=$(tput cols)
ROWS=$(tput lines)

matrix() {
  printf '\033[?25l'  # hide cursor
  while true; do
    col=$((RANDOM % COLS))
    row=$((RANDOM % ROWS))
    char="${CHARS:$((RANDOM % 15)):1}"
    printf "\033[${row};${col}H\033[32m${char}"
  done
}

matrix
